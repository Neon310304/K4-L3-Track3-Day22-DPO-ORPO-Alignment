import time, queue, re, html
from pathlib import Path
from jupyter_client import BlockingKernelClient
connection='/tmp/tmpxamn1vay.json'
k=BlockingKernelClient(connection_file=connection)
k.load_connection_file()
k.start_channels(shell=False,iopub=True,stdin=False,hb=False,control=False)
end=time.monotonic()+70*60
log=Path('/content/day22-clean-t4/submission/evidence/nb3-progress.log')
try:
 with log.open('a',encoding='utf-8') as f:
  f.write('Read-only IOPub subscription to the NB3 kernel; no execution requests sent.\n'); f.flush()
  while time.monotonic()<end and Path(connection).exists():
   try: m=k.get_iopub_msg(timeout=1)
   except queue.Empty: continue
   if m.get('msg_type')=='stream':
    f.write(m.get('content',{}).get('text','')); f.flush()
   elif m.get('msg_type') in ('display_data','update_display_data'):
    display=m.get('content',{}).get('data',{}).get('text/html')
    if display:
     text=html.unescape(re.sub(r'<[^>]*>', ' ', display)); f.write('\nDISPLAY_PROGRESS '+ ' '.join(text.split())[-2000:]+'\n'); f.flush()
   elif m.get('msg_type')=='error':
    c=m.get('content',{}); f.write(str(c.get('ename'))+': '+str(c.get('evalue'))+'\n'); f.flush()
  f.write('\nRead-only observer ended.\n')
finally:
 k.stop_channels()
